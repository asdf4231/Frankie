\section{Scalar Equations (Chapter 24)}

% Section 24.1, pp. 633--635.
\begin{frame}
    \frametitle{Ordinary Differential Equations}
    \begin{block}{\textbf{Definition}}
        An \textbf{ordinary differential equation} is an equation
        \[
            \dot y=F(y,t)
        \]
        between the derivative of an unknown function $y(t)$ and an
        expression $F(y,t)$ involving $y$ and $t$. A solution is a function
        which satisfies that relationship.
    \end{block}
    \begin{itemize}
        \item We write $dy/dt$ as $\dot y$.
        \item If $F$ does not specifically involve $t$, the equation is
        \textbf{autonomous} or \textbf{time-independent}: $\dot y=F(y)$.
        Otherwise, it is \textbf{nonautonomous} or \textbf{time-dependent}.
        \item A \textbf{first order} equation involves only the first
        derivative of the unknown function.
    \end{itemize}
    \medskip
    Reference: Simon and Blume, \emph{Mathematics for Economists},
    Chapters 24--25.
\end{frame}

% Section 24.1, Example 24.5 and the definitions on pp. 636--638.
\begin{frame}
    \frametitle{General Solutions and Initial Values}
    \begin{itemize}
        \item A parameterized solution $y(t,k)$ is a \textbf{general
        solution} if every solution can be achieved by letting $k$ take
        on different values.
        \item The problem of finding a solution which also satisfies
        $y(t_0)=y_0$ is an \textbf{initial value problem}.
    \end{itemize}
    \medskip
    \textbf{Example 24.5.} The amount of money in a bank account with
    interest continuously compounded at annual rate $r$ satisfies
    \[
        \dot y=ry.
    \]
    Its general solution is $y(t)=ke^{rt}$. Knowing the rate at which
    the account grows is not enough to determine its size: we also
    need the original deposit. Since $y(0)=k$, the initial condition
    $y(0)=y_0$ gives
    \[
        y(t)=y_0e^{rt}.
    \]
\end{frame}

% Section 24.2, cases (1)--(2), p. 639.
\begin{frame}
    \frametitle{Linear First Order Equations}
    \begin{enumerate}
        \item $\dot y=ay$, where $a$ is a constant. The general solution is
        \[
            y(t)=ke^{at}.
        \]
        \item $\dot y=ay+b$, where $a$ and $b$ are constants and $a\neq0$.
        The general solution is
        \[
            y(t)=-\frac{b}{a}+ke^{at}.
        \]
    \end{enumerate}
    To verify the second solution, substitute it into the equation:
    \[
        \dot y(t)=ake^{at},\qquad
        ay(t)+b=a\left(-\frac{b}{a}+ke^{at}\right)+b=ake^{at}.
    \]
    \begin{itemize}
        \item $y(t)=-b/a$ is a \textbf{steady state solution},
        corresponding to $k=0$.
        \item Without the $b$-term the equation is \textbf{homogeneous};
        with the $b$-term it is \textbf{nonhomogeneous}.
    \end{itemize}
\end{frame}

% Section 24.2, cases (3)--(4), pp. 639--640.
% Definite lower limits make the initial condition explicit; a(t), b(t)
% and the integrating-factor calculation follow the textbook.
\begin{frame}
    \frametitle{Integrating Factors}
    Consider the nonautonomous linear equation
    \[
        \dot y=a(t)y+b(t).
    \]
    Write it as $\dot y-a(t)y=b(t)$ and multiply by
    $\exp\left(-\int_{t_0}^t a(s)\,ds\right)$:
    \[
        \frac{d}{dt}\left[y(t)e^{-\int_{t_0}^t a(s)\,ds}\right]
        =b(t)e^{-\int_{t_0}^t a(s)\,ds}.
    \]
    The expression which makes the left side an exact derivative is
    called an \textbf{integrating factor}. Integrating gives
    \[
        y(t)=\left[k+\int_{t_0}^t
        b(s)e^{-\int_{t_0}^s a(u)\,du}\,ds\right]
        e^{\int_{t_0}^t a(s)\,ds},\qquad k=y(t_0).
    \]
    For the homogeneous equation $\dot y=a(t)y$, this reduces to
    \[
        y(t)=ke^{\int_{t_0}^t a(s)\,ds}.
    \]
\end{frame}

% Section 24.2, pp. 641--644; Example 24.10.
\begin{frame}
    \frametitle{Separable Equations}
    An equation $\dot y=F(y,t)$ is \textbf{separable} if
    $F(y,t)=g(y)h(t)$. On an interval where $g(y)\neq0$, write
    \[
        \frac{dy}{g(y)}=h(t)\,dt,\qquad
        \int\frac{dy}{g(y)}=\int h(t)\,dt+c.
    \]
    \medskip
    \textbf{Example 24.10.} Consider the autonomous equation
    $\dot y=y^2$. Separating and integrating gives
    \[
        \int y^{-2}\,dy=\int dt+c,\qquad
        -y^{-1}=t+c.
    \]
    Thus the nonzero solutions are
    \[
        y(t)=\frac{1}{k-t},\qquad k=-c.
    \]
    Constant solutions with $g(y)=0$ must be checked separately. Here
    $y(t)=0$ is also a solution.
\end{frame}

% Section 24.4, Theorem 24.5, p. 658.
% Neighborhood regularity is made explicit, as in Section 25.1, p. 677.
\begin{frame}
    \frametitle{Existence and Uniqueness}
    \begin{block}{\textbf{Theorem 24.5}}
        Consider the initial value problem
        \[
            \dot y=f(t,y),\qquad y(t_0)=y_0.
        \]
        Suppose that $f$ is continuous in a neighborhood of $(t_0,y_0)$.
        Then there exists a $C^1$ function $y:I\rightarrow\RR$ defined
        on an open interval $I=(t_0-\alpha,t_0+\alpha)$ such that
        \[
            y(t_0)=y_0,\qquad \dot y(t)=f(t,y(t))\quad\text{for all }t\in I.
        \]
        Furthermore, if $f$ is $C^1$ in that neighborhood, the local
        solution is unique.
    \end{block}
    \begin{itemize}
        \item A solution may exist even when we cannot write it in a
        closed form.
        \item The result is local: it gives a solution on an interval
        about $t_0$, not necessarily for all future time.
    \end{itemize}
\end{frame}

% Sections 24.4--24.5, Example 24.20 and Figure 24.11, pp. 663--666.
\begin{frame}
    \frametitle{Phase Portraits on the Line}
    For an autonomous equation $\dot y=f(y)$, the evolution depends on
    where the process starts, not on when it starts.
    \begin{columns}[c]
        \begin{column}{0.49\textwidth}
            \begin{itemize}
                \item Constant solutions are called \textbf{steady
                states}, \textbf{rest points}, or \textbf{equilibria}.
                They satisfy $f(y)=0$.
                \item To draw a \textbf{phase portrait}, find the zeros
                of $f$ and check its sign between the zeros.
                \item If $f(y)>0$, $y(t)$ is increasing; if $f(y)<0$,
                $y(t)$ is decreasing.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \centering
            \includegraphics[width=\linewidth]{figures/lecture09/figure24-11.png}

            {\footnotesize Simon and Blume, Figure 24.11.}
        \end{column}
    \end{columns}
    \medskip
    For $\dot y=y(2-y)$, every solution with $y(0)>0$ tends to $y=2$.
    The equilibrium $y=0$ is unstable.
\end{frame}

% Section 24.5, Theorem 24.6 and its preceding argument, pp. 668--669.
\begin{frame}
    \frametitle{Stability of Equilibria on the Line}
    \begin{block}{\textbf{Theorem 24.6}}
        Let $y_0$ be a rest point of the $C^1$ differential equation
        $\dot y=f(y)$ on the line, so $f(y_0)=0$.
        \begin{itemize}
            \item If $f'(y_0)<0$, then $y_0$ is an asymptotically stable
            equilibrium.
            \item If $f'(y_0)>0$, then $y_0$ is an unstable equilibrium.
        \end{itemize}
    \end{block}
    \textbf{Proof sketch:} If $f'(y_0)<0$, $f$ is decreasing near $y_0$.
    Since $f(y_0)=0$, $f(y)>0$ on its left and $f(y)<0$ on its right.
    The flow moves toward $y_0$ on both sides. If $f'(y_0)>0$, the signs
    are reversed and the flow moves away from $y_0$.
    \begin{itemize}
        \item For $f(y)=y(2-y)$, $f'(0)=2>0$ and $f'(2)=-2<0$.
        \item If $f'(y_0)=0$, the test does not determine stability;
        more information is needed.
    \end{itemize}
\end{frame}

\section{Systems of Equations (Chapter 25)}

% Section 25.1, pp. 676--677.
\begin{frame}
    \frametitle{Planar Systems}
    The general first order system of two differential equations is
    \[
        \dot x=F(x,y,t),\qquad \dot y=G(x,y,t).
    \]
    \begin{itemize}
        \item A solution is a pair of functions $x^*(t)$ and $y^*(t)$
        satisfying both equations at every $t$ in their domain.
        \item A general solution contains two independent parameters.
        To specify a particular solution, prescribe an initial condition
        for each variable:
        \[
            x(t_0)=x_0,\qquad y(t_0)=y_0.
        \]
        \item If $F$ and $G$ do not depend explicitly on $t$, the system
        is \textbf{autonomous}. We will work with autonomous systems.
        \item The existence and uniqueness result also holds for
        systems: continuous $F$ and $G$ give local existence; $C^1$
        functions give uniqueness.
    \end{itemize}
\end{frame}

% Sections 23.1 and 23.3; restricted to the two-dimensional prerequisite.
\begin{frame}
    \frametitle{Eigenvalues and Eigenvectors}
    \begin{block}{\textbf{Definition}}
        A number $r$ is an \textbf{eigenvalue} of a square matrix $A$ if
        there is a nonzero vector $\mathbf{v}$ such that
        \[
            A\mathbf{v}=r\mathbf{v}.
        \]
        The vector $\mathbf{v}$ is an \textbf{eigenvector} corresponding
        to $r$.
    \end{block}
    Eigenvalues solve the \textbf{characteristic equation}
    $\det(A-rI)=0$. For a $2\times2$ matrix,
    \[
        A=\begin{pmatrix}a&b\\c&d\end{pmatrix},\qquad
        \det(A-rI)=r^2-(a+d)r+(ad-bc).
    \]
    Its roots $r_1,r_2$ satisfy
    \[
        r_1+r_2=\operatorname{trace}A=a+d,\qquad
        r_1r_2=\det A=ad-bc.
    \]
    Find an eigenvector by solving $(A-rI)\mathbf{v}=\mathbf{0}$.
\end{frame}

% Section 25.2, Theorem 25.1 and its derivation, pp. 678--680.
\begin{frame}
    \frametitle{Linear Systems via Eigenvalues}
    \begin{block}{\textbf{Theorem 25.1 (Two-Dimensional Case)}}
        Suppose that the $2\times2$ matrix $A$ has distinct real
        eigenvalues $r_1,r_2$, with corresponding eigenvectors
        $\mathbf{v}_1,\mathbf{v}_2$. The general solution of
        $\dot{\mathbf{x}}=A\mathbf{x}$ is
        \[
            \mathbf{x}(t)=c_1e^{r_1t}\mathbf{v}_1
                         +c_2e^{r_2t}\mathbf{v}_2.
        \]
    \end{block}
    \textbf{Proof:} Eigenvectors for distinct eigenvalues are linearly
    independent. Set $P=[\mathbf{v}_1\ \mathbf{v}_2]$ and
    $D=\operatorname{diag}(r_1,r_2)$. Then $AP=PD$ and
    \[
        \mathbf{y}=P^{-1}\mathbf{x}\quad\Longrightarrow\quad
        \dot{\mathbf{y}}=P^{-1}AP\mathbf{y}=D\mathbf{y}.
    \]
    Thus $y_i(t)=c_ie^{r_it}$. Returning to $\mathbf{x}=P\mathbf{y}$
    gives the stated solution. The initial condition determines $c_1,c_2$.
\end{frame}

% Section 25.4, pp. 684--686; Exercise 25.18 asks for formal definitions.
\begin{frame}
    \frametitle{Steady States and Their Stability}
    For the autonomous system
    \[
        \dot{\mathbf{y}}=F(\mathbf{y}),
    \]
    a point $\mathbf{y}^*$ is a \textbf{steady state} if
    $F(\mathbf{y}^*)=\mathbf{0}$. Finding steady states amounts to
    solving a system of algebraic equations.
    \begin{itemize}
        \item A steady state is \textbf{stable} if solutions starting
        sufficiently close remain close for all future time.
        \item It is \textbf{asymptotically stable} if it is stable and
        every solution starting sufficiently near it converges to it
        as $t\rightarrow\infty$.
        \item If a steady state is not stable, it is \textbf{unstable}.
        \item Stability concerns nearby solutions, not just the constant
        solution at the equilibrium itself.
    \end{itemize}
\end{frame}

% Section 25.4, Theorem 25.4(a)--(b), pp. 686--687.
\begin{frame}
    \frametitle{Stability of Linear Systems}
    \begin{block}{\textbf{Theorem 25.4 (Parts (a)--(b))}}
        The constant solution $\mathbf{x}=\mathbf{0}$ is always a
        steady state of $\dot{\mathbf{x}}=A\mathbf{x}$.
        \begin{enumerate}
            \item[(a)] If every eigenvalue of $A$ has negative real
            part, the origin is globally asymptotically stable:
            every solution tends to $\mathbf{0}$ as $t\rightarrow\infty$.
            \item[(b)] If $A$ has an eigenvalue with positive real part,
            the origin is unstable.
        \end{enumerate}
    \end{block}
    \textbf{Explanation:} With distinct real eigenvalues, solutions
    are sums of terms $c_ie^{r_it}\mathbf{v}_i$.
    \begin{itemize}
        \item If every $r_i<0$, all terms tend to zero.
        \item If some $r_i>0$, a nonzero component in its eigenvector
        direction grows without bound.
        \item For complex eigenvalues, the real part determines the
        exponential growth or decay of the oscillating terms.
    \end{itemize}
\end{frame}

% Section 25.5, pp. 690--692.
\begin{frame}
    \frametitle{Vector Fields and Phase Portraits}
    Consider the planar system
    \[
        \dot x=f(x,y),\qquad \dot y=g(x,y).
    \]
    \begin{itemize}
        \item At each point $(x,y)$, the velocity vector is
        $(f(x,y),g(x,y))$. The family of these vectors is called a
        \textbf{vector field}.
        \item A solution $(x(t),y(t))$ is a parameterized curve in the
        plane which is everywhere tangent to the vector field.
        \item The set of these solution curves is the \textbf{phase
        portrait}, or \textbf{phase diagram}, of the system.
        \item The curves describe the paths or \textbf{orbits} of the
        system. Arrows indicate the direction of increasing $t$.
    \end{itemize}
\end{frame}

            \textbf{Example 25.7.} Consider
            \[
                \dot x=2x,\qquad \dot y=-2y.

            Its general solution is

% Consequence of Theorem 25.1, solution (10), and Theorem 25.4.
\begin{frame}
    \frametitle{Linear Systems: Convergent Solutions}
    Suppose that $r_1<0<r_2$. The general solution is
    \[
        \mathbf{x}(t)=c_1e^{r_1t}\mathbf{v}_1
                     +c_2e^{r_2t}\mathbf{v}_2.
    \]
    \begin{itemize}
        \item The component in the $\mathbf{v}_1$ direction tends to
        zero; a nonzero component in the $\mathbf{v}_2$ direction grows.
        \item Therefore, a solution converges to the origin if and only
        if $c_2=0$:
        \[
            \mathbf{x}(t)=c_1e^{r_1t}\mathbf{v}_1.
        \]
        \item The convergent solutions lie on the line through the
        eigenvector associated with the negative eigenvalue.
        \item This is the geometry of a \textbf{saddle}: some solutions
        converge to the equilibrium even though it is unstable.
    \end{itemize}
\end{frame}

% Section 25.8, p. 715. This is the textbook's linearization sketch,
% not a full proof of the nonlinear stability theorem.
\begin{frame}
    \frametitle{Linearization}
    Consider $\dot{\mathbf{x}}=F(\mathbf{x})$ near a steady state
    $\mathbf{x}^*$, where $F(\mathbf{x}^*)=\mathbf{0}$.
    Assume $F$ is $C^1$ near $\mathbf{x}^*$.
    Put $\mathbf{h}(t)=\mathbf{x}(t)-\mathbf{x}^*$. Taylor expansion gives
    \begin{align*}
        \dot{\mathbf{h}}(t)
        &=F(\mathbf{x}^*+\mathbf{h}(t))\\
        &=F(\mathbf{x}^*)+DF(\mathbf{x}^*)\mathbf{h}(t)+R(\mathbf{h}(t))\\
        &=DF(\mathbf{x}^*)\mathbf{h}(t)+R(\mathbf{h}(t)),
    \end{align*}
    where
    \[
        \frac{\|R(\mathbf{h})\|}{\|\mathbf{h}\|}\longrightarrow0
        \quad\text{as }\mathbf{h}\longrightarrow\mathbf{0}.
    \]
    The linear system
    \[
        \dot{\mathbf{h}}=DF(\mathbf{x}^*)\mathbf{h}
    \]
    is the \textbf{linearization} around $\mathbf{x}^*$. The next theorem
    states when it determines the local stability of the nonlinear system.
\end{frame}

% Section 25.4, Theorem 25.5, p. 687.
\begin{frame}
    \frametitle{Stability of Nonlinear Systems}
    \begin{block}{\textbf{Theorem 25.5}}
        Let $\mathbf{y}^*$ be a steady state of
        $\dot{\mathbf{y}}=F(\mathbf{y})$ on $\RR^n$, where
        $F:\RR^n\rightarrow\RR^n$ is $C^1$.
        \begin{enumerate}
            \item[(a)] If each eigenvalue of the Jacobian matrix
            $DF(\mathbf{y}^*)$ is negative or has negative real part,
            then $\mathbf{y}^*$ is asymptotically stable.
            \item[(b)] If $DF(\mathbf{y}^*)$ has at least one positive
            real eigenvalue or one complex eigenvalue with positive
            real part, then $\mathbf{y}^*$ is unstable.
        \end{enumerate}
    \end{block}
    \begin{itemize}
        \item The Jacobian replaces the derivative in the one-dimensional
        stability test.
        \item If there are zero or purely imaginary eigenvalues and none
        with positive real part, this test does not determine stability.
        \item This is analogous to an inconclusive second derivative
        test in optimization.
    \end{itemize}
\end{frame}

% Exercises 25.11 and 25.13, p. 689, using Section 23.3.
\begin{frame}
    \frametitle{Trace and Determinant}
    For a planar system, let $A$ be the Jacobian at a steady state.
    Its characteristic equation is
    \[
        r^2-(\operatorname{trace}A)r+\det A=0.
    \]
    \begin{itemize}
        \item If $\det A<0$, the eigenvalues are real and have opposite
        signs: a saddle configuration.
        \item If $\det A>0$ and $\operatorname{trace}A<0$, both
        eigenvalues have negative real part: asymptotic stability.
        \item If $\det A>0$ and $\operatorname{trace}A>0$, both
        eigenvalues have positive real part: instability.
    \end{itemize}
    \textbf{Explanation:} The roots have product $\det A$ and sum
    $\operatorname{trace}A$. Real roots with positive product have the
    same sign; a complex conjugate pair has common real part
    $\operatorname{trace}A/2$. A negative product gives real roots of
    opposite signs.
    \medskip

    These are the criteria developed in Exercises 25.11 and 25.13.
\end{frame}

% Section 25.5, Steps 1--5, pp. 694--700.
\begin{frame}
    \frametitle{Drawing Phase Portraits}
    For $\dot x=f(x,y)$ and $\dot y=g(x,y)$:
    \begin{enumerate}
        \item Find the equilibria by solving $f(x,y)=g(x,y)=0$.
        \item Use the Jacobian to determine their local stability.
        \item Draw the \textbf{isoclines}: the curves $f(x,y)=0$ and
        $g(x,y)=0$, where the vector field is vertical or horizontal.
        \item Fill in the arrows on the isoclines and in the sectors
        between them. Evaluate the signs of $f$ and $g$ at a point in
        each sector.
        \item Sketch representative solution curves following these
        directions.
    \end{enumerate}
    \medskip
    The isoclines divide the plane into regions in which the vector field
    points northeast, northwest, southeast, or southwest. At an
    equilibrium the vector field is zero.
\end{frame}

% Section 25.4, Example 25.2, p. 688.
\begin{frame}
    \frametitle{Example 25.2: Competing Species}
    Consider the system
    \[
        \dot x=x(4-x-y),\qquad \dot y=y(6-y-3x).
    \]
    Its Jacobian is
    \[
        D(f,g)(x,y)=
        \begin{pmatrix}
            4-2x-y & -x\\
            -3y & 6-2y-3x
        \end{pmatrix}.
    \]
    The equilibria and their stability are:
    \begin{center}
        \begin{tabular}{lll}
            \toprule
            Equilibrium & Eigenvalues & Stability\\
            \midrule
            $(0,0)$ & $4,\ 6$ & Unstable\\
            $(0,6)$ & $-2,\ -6$ & Asymptotically stable\\
            $(4,0)$ & $-4,\ -6$ & Asymptotically stable\\
            $(1,3)$ & Roots of $r^2+4r-6=0$ & Unstable (opposite signs)\\
            \bottomrule
        \end{tabular}
    \end{center}
    At $(1,3)$, the product of the eigenvalues is $-6$, so one is positive
    and the other negative.
\end{frame}

% Section 25.5, system (32) and its phase portrait, pp. 694--702.
\begin{frame}
    \frametitle{Competing Species: Phase Portrait}
    \begin{columns}[c]
        \begin{column}{0.49\textwidth}
            The isoclines are
            \begin{align*}
                \dot x=0 &: \quad x=0\ \text{or}\ 4-x-y=0,\\
                \dot y=0 &: \quad y=0\ \text{or}\ 6-y-3x=0.
            \end{align*}
            \begin{itemize}
                \item Different initial populations can lead to
                convergence to $(0,6)$ or $(4,0)$.
                \item The dividing curve, or \textbf{separatrix},
                consists of trajectories tending to $(1,3)$ and the
                equilibrium itself.
                \item An exogenous shock which moves the system across
                this curve changes its long-run behavior.
            \end{itemize}
        \end{column}
